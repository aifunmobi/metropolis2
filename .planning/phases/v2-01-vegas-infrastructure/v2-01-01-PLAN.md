```javascript
function mkVegasHighway() {
    const highwayGroup = new THREE.Group();

    // Highway road surface - dark asphalt
    const roadMat = new THREE.MeshStandardMaterial({
        color: 0x333333,
        roughness: 0.9
    });

    // Main highway extending from Metropolis east edge to Vegas zone
    // Starts at halfSize (Metropolis edge) and extends to Vegas
    const startX = halfSize;
    const endX = VEGAS_CFG.zoneStartX + VEGAS_CFG.highwayLength;
    const highwayLength = endX - startX;

    const highway = new THREE.Mesh(
        new THREE.PlaneGeometry(highwayLength, VEGAS_CFG.highwayWidth),
        roadMat
    );
    highway.rotation.x = -Math.PI / 2;
    highway.position.set(startX + highwayLength / 2, 0.02, 0); // Centered on Z=0
    highwayGroup.add(highway);

    // Lane markings - white dashed center line
    const lineMat = new THREE.MeshStandardMaterial({ color: 0xffffff });
    const dashLength = 3;
    const gapLength = 2;
    for (let x = startX + 2; x < endX - 2; x += dashLength + gapLength) {
        const dash = new THREE.Mesh(
            new THREE.PlaneGeometry(dashLength, 0.3),
            lineMat
        );
        dash.rotation.x = -Math.PI / 2;
        dash.position.set(x + dashLength / 2, 0.03, 0);
        highwayGroup.add(dash);
    }

    // Edge lines - solid white
    const edgeY = 0.03;
    const edgeOffset = VEGAS_CFG.highwayWidth / 2 - 0.5;
    for (const side of [-1, 1]) {
        const edge = new THREE.Mesh(
            new THREE.PlaneGeometry(highwayLength - 4, 0.25),
            lineMat
        );
        edge.rotation.x = -Math.PI / 2;
        edge.position.set(startX + highwayLength / 2, edgeY, side * edgeOffset);
        highwayGroup.add(edge);
    }

    // Desert ground alongside highway
    const desertMat = new THREE.MeshStandardMaterial({
        color: 0xc2956e, // Sandy desert color
        roughness: 0.95
    });

    // Desert strips on both sides of highway
    const desertWidth = 25;
    for (const side of [-1, 1]) {
        const desert = new THREE.Mesh(
            new THREE.PlaneGeometry(highwayLength + 20, desertWidth),
            desertMat
        );
        desert.rotation.x = -Math.PI / 2;
        desert.position.set(
            startX + highwayLength / 2,
            0.005,
            side * (VEGAS_CFG.highwayWidth / 2 + desertWidth / 2)
        );
        highwayGroup.add(desert);
    }

    scene.add(highwayGroup);
    return highwayGroup;
}
```

```javascript
function mkRoute66Sign() {
    const sign = new THREE.Group();

    // Sign post - weathered wood or metal
    const postMat = new THREE.MeshStandardMaterial({
        color: 0x665544,
        roughness: 0.8
    });
    const post = new THREE.Mesh(
        new THREE.CylinderGeometry(0.15, 0.18, 4, 8),
        postMat
    );
    post.position.y = 2;
    sign.add(post);

    // Shield-shaped sign using canvas texture
    // Route 66 signs have distinctive shield shape
    const canvas = document.createElement('canvas');
    canvas.width = 256;
    canvas.height = 256;
    const ctx = canvas.getContext('2d');

    // White background
    ctx.fillStyle = '#ffffff';
    ctx.fillRect(0, 0, 256, 256);

    // Black border/shield outline
    ctx.strokeStyle = '#000000';
    ctx.lineWidth = 8;
    ctx.beginPath();
    ctx.moveTo(128, 20);      // Top center
    ctx.lineTo(220, 60);      // Top right
    ctx.lineTo(220, 200);     // Bottom right
    ctx.lineTo(128, 240);     // Bottom center
    ctx.lineTo(36, 200);      // Bottom left
    ctx.lineTo(36, 60);       // Top left
    ctx.closePath();
    ctx.stroke();
    ctx.fillStyle = '#ffffff';
    ctx.fill();

    // "ROUTE" text
    ctx.fillStyle = '#000000';
    ctx.font = 'bold 28px Arial';
    ctx.textAlign = 'center';
    ctx.fillText('ROUTE', 128, 70);

    // "66" large text
    ctx.font = 'bold 80px Arial';
    ctx.fillText('66', 128, 160);

    // State text (optional)
    ctx.font = '18px Arial';
    ctx.fillText('US', 128, 210);

    const texture = new THREE.CanvasTexture(canvas);
    const signPlane = new THREE.Mesh(
        new THREE.PlaneGeometry(2, 2),
        new THREE.MeshBasicMaterial({ map: texture, transparent: true })
    );
    signPlane.position.y = 4.5;
    sign.add(signPlane);

    // Back side of sign (same texture, rotated)
    const signBack = signPlane.clone();
    signBack.rotation.y = Math.PI;
    signBack.position.z = -0.05;
    sign.add(signBack);

    // Position sign near highway start (just past Metropolis edge)
    sign.position.set(halfSize + 5, 0, -VEGAS_CFG.highwayWidth / 2 - 3);

    scene.add(sign);
    return sign;
}
```
