function mkStreetlights() {
    const poleHeight = 6;
    const armLength = 2;
    const poleMat = new THREE.MeshStandardMaterial({ color: 0x444444, metalness: 0.8 });
    const bulbMat = new THREE.MeshBasicMaterial({ color: 0x222211 }); // Off by default

    // Place streetlights along all roads
    for (let i = 0; i <= GRID.gridCount; i++) {
        const roadPos = -halfSize + i * GRID.cellSize;

        // Horizontal roads (along X axis)
        for (let x = -halfSize + 8; x < halfSize - 4; x += 16) {
            // Skip intersection areas
            const nearIntersection = Array.from({length: GRID.gridCount + 1}, (_, j) =>
                -halfSize + j * GRID.cellSize
            ).some(iPos => Math.abs(x - iPos) < GRID.roadWidth);

            if (!nearIntersection) {
                // North side of road
                mkStreetlight(x, roadPos - GRID.roadWidth/2 - 1, 0);
                // South side of road
                mkStreetlight(x, roadPos + GRID.roadWidth/2 + 1, Math.PI);
            }
        }

        // Vertical roads (along Z axis)
        for (let z = -halfSize + 8; z < halfSize - 4; z += 16) {
            const nearIntersection = Array.from({length: GRID.gridCount + 1}, (_, j) =>
                -halfSize + j * GRID.cellSize
            ).some(iPos => Math.abs(z - iPos) < GRID.roadWidth);

            if (!nearIntersection) {
                // East side of road
                mkStreetlight(roadPos + GRID.roadWidth/2 + 1, z, -Math.PI/2);
                // West side of road
                mkStreetlight(roadPos - GRID.roadWidth/2 - 1, z, Math.PI/2);
            }
        }
    }

    function mkStreetlight(x, z, rotation) {
        const g = new THREE.Group();

        // Pole
        const pole = new THREE.Mesh(
            new THREE.CylinderGeometry(0.12, 0.15, poleHeight, 8),
            poleMat
        );
        pole.position.y = poleHeight / 2;
        g.add(pole);

        // Curved arm
        const arm = new THREE.Mesh(
            new THREE.BoxGeometry(armLength, 0.1, 0.1),
            poleMat
        );
        arm.position.set(armLength/2, poleHeight - 0.1, 0);
        g.add(arm);

        // Light housing
        const housing = new THREE.Mesh(
            new THREE.CylinderGeometry(0.3, 0.4, 0.3, 8),
            poleMat
        );
        housing.position.set(armLength, poleHeight - 0.3, 0);
        g.add(housing);

        // Bulb (visual)
        const bulb = new THREE.Mesh(
            new THREE.SphereGeometry(0.25, 8, 8),
            bulbMat.clone() // Clone so each can be controlled independently
        );
        bulb.position.set(armLength, poleHeight - 0.5, 0);
        g.add(bulb);
        g.userData.bulb = bulb;

        // Point light (actual light emission) - starts off
        const light = new THREE.PointLight(0xffeecc, 0, 15);
        light.position.set(armLength, poleHeight - 0.6, 0);
        g.add(light);
        g.userData.light = light;

        g.position.set(x, 0, z);
        g.rotation.y = rotation;

        streetlights.push(g);
        scene.add(g);
    }
}

function updStreetlights() {
    const targetIntensity = isNight ? 1.5 : 0;
    const targetColor = isNight ? 0xffeeaa : 0x222211;

    streetlights.forEach(sl => {
        const light = sl.userData.light;
        const bulb = sl.userData.bulb;

        if (light) {
            // Smooth transition
            light.intensity += (targetIntensity - light.intensity) * 0.05;
        }

        if (bulb) {
            bulb.material.color.setHex(targetColor);
        }
    });
}

Find updWindowLights() (around line 5239). The current logic randomly toggles windows. We need to bias it toward more lit windows at night:

```javascript
function updWindowLights(t) {
    if (t >= lastWindowToggle) {
        if (allWindows.length > 0) {
            // More activity at night, less during day
            const numToToggle = 5 + Math.floor(Math.random() * 6);

            for (let i = 0; i < numToToggle; i++) {
                const win = allWindows[Math.floor(Math.random() * allWindows.length)];

                // Probability of being lit depends on time
                // Night (isNight=true): 70% chance lit
                // Day (isNight=false): 20% chance lit
                const litProbability = isNight ? 0.7 : 0.2;

                // Instead of just toggling, set based on probability
                const shouldBeLit = Math.random() < litProbability;

                if (shouldBeLit !== win.userData.isLit) {
                    win.userData.isLit = shouldBeLit;
                    win.material.color.setHex(win.userData.isLit ? 0xffeeaa : 0x334455);
                    win.material.needsUpdate = true;
                }
            }
        }
        lastWindowToggle = t + 0.5 + Math.random() * 1;
    }
}
```
