function mkBird() {
    const g = new THREE.Group();

    // Body - small elongated ellipsoid
    const body = new THREE.Mesh(
        new THREE.SphereGeometry(0.15, 8, 6),
        new THREE.MeshStandardMaterial({ color: 0x333333 })
    );
    body.scale.set(1, 0.6, 1.5);
    g.add(body);

    // Head
    const head = new THREE.Mesh(
        new THREE.SphereGeometry(0.08, 6, 6),
        new THREE.MeshStandardMaterial({ color: 0x222222 })
    );
    head.position.set(0, 0.05, 0.2);
    g.add(head);

    // Beak
    const beak = new THREE.Mesh(
        new THREE.ConeGeometry(0.03, 0.1, 4),
        new THREE.MeshStandardMaterial({ color: 0xff8800 })
    );
    beak.rotation.x = -Math.PI / 2;
    beak.position.set(0, 0.03, 0.3);
    g.add(beak);

    // Wings - simple triangular shapes
    const wingGeo = new THREE.BufferGeometry();
    const wingVerts = new Float32Array([
        0, 0, 0,      // origin
        0.4, 0, 0.1,  // wingtip
        0.1, 0, 0.15  // trailing edge
    ]);
    wingGeo.setAttribute('position', new THREE.BufferAttribute(wingVerts, 3));
    wingGeo.computeVertexNormals();
    const wingMat = new THREE.MeshStandardMaterial({
        color: 0x444444,
        side: THREE.DoubleSide
    });

    const leftWing = new THREE.Mesh(wingGeo, wingMat);
    leftWing.position.set(0.05, 0, 0);
    g.add(leftWing);
    g.userData.leftWing = leftWing;

    const rightWing = new THREE.Mesh(wingGeo.clone(), wingMat);
    rightWing.scale.x = -1;
    rightWing.position.set(-0.05, 0, 0);
    g.add(rightWing);
    g.userData.rightWing = rightWing;

    // Initialize bird userData
    g.userData.velocity = new THREE.Vector3(
        (Math.random() - 0.5) * 0.2,
        0,
        (Math.random() - 0.5) * 0.2
    );
    g.userData.wingPhase = Math.random() * Math.PI * 2;
    g.userData.state = 'flying';  // 'flying', 'landing', 'landed', 'takeoff'

    return g;
}

function mkFlock(size, startPos) {
    const flock = {
        birds: [],
        center: startPos.clone(),
        targetWaypoint: null
    };

    for (let i = 0; i < size; i++) {
        const bird = mkBird();
        // Scatter birds around start position
        bird.position.set(
            startPos.x + (Math.random() - 0.5) * 8,
            startPos.y + (Math.random() - 0.5) * 4,
            startPos.z + (Math.random() - 0.5) * 8
        );
        flock.birds.push(bird);
        scene.add(bird);
    }

    // Set initial waypoint
    flock.targetWaypoint = getRandomWaypoint();

    return flock;
}

function getRandomWaypoint() {
    const margin = BIRD_CFG.boundaryMargin;
    return new THREE.Vector3(
        (Math.random() - 0.5) * (halfSize * 2 + margin),
        BIRD_CFG.minHeight + Math.random() * (BIRD_CFG.maxHeight - BIRD_CFG.minHeight),
        (Math.random() - 0.5) * (halfSize * 2 + margin)
    );
}

function mkFlocks() {
    for (let i = 0; i < CFG.flocks; i++) {
        const size = BIRD_CFG.flockSizes[i % BIRD_CFG.flockSizes.length];
        const startPos = new THREE.Vector3(
            (Math.random() - 0.5) * halfSize * 1.5,
            BIRD_CFG.minHeight + Math.random() * 20,
            (Math.random() - 0.5) * halfSize * 1.5
        );
        const flock = mkFlock(size, startPos);
        flocks.push(flock);
    }
}

function updBirds(t) {
    flocks.forEach(flock => {
        // Update flock center
        flock.center.set(0, 0, 0);
        flock.birds.forEach(bird => {
            if (bird.userData.state === 'flying') {
                flock.center.add(bird.position);
            }
        });
        const flyingCount = flock.birds.filter(b => b.userData.state === 'flying').length;
        if (flyingCount > 0) {
            flock.center.divideScalar(flyingCount);
        }

        // Check if flock reached waypoint, pick new one
        if (flock.targetWaypoint && flock.center.distanceTo(flock.targetWaypoint) < 15) {
            flock.targetWaypoint = getRandomWaypoint();
        }

        // Update each bird
        flock.birds.forEach(bird => {
            if (bird.userData.state !== 'flying') return;

            // Calculate boids forces
            const separation = new THREE.Vector3();
            const cohesion = new THREE.Vector3();
            const alignment = new THREE.Vector3();
            let separationCount = 0;
            let cohesionCount = 0;
            let alignmentCount = 0;

            // Check against other birds in same flock
            flock.birds.forEach(other => {
                if (other === bird || other.userData.state !== 'flying') return;

                const dist = bird.position.distanceTo(other.position);

                // Separation - steer away from nearby birds
                if (dist < BIRD_CFG.separationDist && dist > 0) {
                    const diff = bird.position.clone().sub(other.position);
                    diff.normalize().divideScalar(dist);
                    separation.add(diff);
                    separationCount++;
                }

                // Cohesion - steer toward center of nearby birds
                if (dist < BIRD_CFG.cohesionDist) {
                    cohesion.add(other.position);
                    cohesionCount++;
                }

                // Alignment - match velocity of nearby birds
                if (dist < BIRD_CFG.alignmentDist) {
                    alignment.add(other.userData.velocity);
                    alignmentCount++;
                }
            });

            // Average and apply weights
            if (separationCount > 0) {
                separation.divideScalar(separationCount);
                separation.normalize().multiplyScalar(BIRD_CFG.separationWeight);
            }

            if (cohesionCount > 0) {
                cohesion.divideScalar(cohesionCount);
                cohesion.sub(bird.position);
                cohesion.normalize().multiplyScalar(BIRD_CFG.cohesionWeight);
            }

            if (alignmentCount > 0) {
                alignment.divideScalar(alignmentCount);
                alignment.normalize().multiplyScalar(BIRD_CFG.alignmentWeight);
            }

            // Waypoint steering
            const toWaypoint = new THREE.Vector3();
            if (flock.targetWaypoint) {
                toWaypoint.subVectors(flock.targetWaypoint, bird.position);
                toWaypoint.normalize().multiplyScalar(0.5);
            }

            // Random wandering
            const wander = new THREE.Vector3(
                (Math.random() - 0.5) * BIRD_CFG.wanderWeight,
                (Math.random() - 0.5) * BIRD_CFG.wanderWeight * 0.3,
                (Math.random() - 0.5) * BIRD_CFG.wanderWeight
            );

            // Height bounds steering
            const heightSteer = new THREE.Vector3();
            if (bird.position.y < BIRD_CFG.minHeight) {
                heightSteer.y = 0.5;
            } else if (bird.position.y > BIRD_CFG.maxHeight) {
                heightSteer.y = -0.5;
            }

            // Combine all forces
            const acceleration = new THREE.Vector3();
            acceleration.add(separation);
            acceleration.add(cohesion);
            acceleration.add(alignment);
            acceleration.add(toWaypoint);
            acceleration.add(wander);
            acceleration.add(heightSteer);

            // Apply acceleration to velocity
            bird.userData.velocity.add(acceleration.multiplyScalar(BIRD_CFG.turnSpeed));

            // Limit speed
            const speed = bird.userData.velocity.length();
            if (speed > BIRD_CFG.speed * 1.5) {
                bird.userData.velocity.normalize().multiplyScalar(BIRD_CFG.speed * 1.5);
            } else if (speed < BIRD_CFG.speed * 0.5) {
                bird.userData.velocity.normalize().multiplyScalar(BIRD_CFG.speed * 0.5);
            }

            // Update position
            bird.position.add(bird.userData.velocity);

            // Face direction of travel
            if (bird.userData.velocity.length() > 0.01) {
                const lookTarget = bird.position.clone().add(bird.userData.velocity);
                bird.lookAt(lookTarget);
            }

            // Animate wings
            bird.userData.wingPhase += 0.4;
            const wingAngle = Math.sin(bird.userData.wingPhase) * 0.5;
            if (bird.userData.leftWing) {
                bird.userData.leftWing.rotation.z = wingAngle;
            }
            if (bird.userData.rightWing) {
                bird.userData.rightWing.rotation.z = -wingAngle;
            }
        });
    });
}
