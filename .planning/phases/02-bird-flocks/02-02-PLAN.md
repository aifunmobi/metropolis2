```javascript
function findPerchSpot(bird) {
    // 70% chance rooftop, 30% chance sidewalk
    const preferRooftop = Math.random() < 0.7;

    if (preferRooftop) {
        // Find a nearby building rooftop
        let bestBuilding = null;
        let bestDist = Infinity;

        buildings.forEach(b => {
            if (!b.userData || !b.userData.height) return;
            const dist = bird.position.distanceTo(b.position);
            if (dist < 40 && dist < bestDist) {
                bestDist = dist;
                bestBuilding = b;
            }
        });

        if (bestBuilding) {
            const h = bestBuilding.userData.height || 10;
            const w = bestBuilding.userData.width || 8;
            // Random spot on rooftop
            return new THREE.Vector3(
                bestBuilding.position.x + (Math.random() - 0.5) * w * 0.6,
                h + 0.1,
                bestBuilding.position.z + (Math.random() - 0.5) * w * 0.6
            );
        }
    }

    // Sidewalk perch - find a spot near bird's current position
    const innerSize = GRID.cellSize - GRID.roadWidth;
    const bx = Math.floor((bird.position.x + halfSize) / GRID.cellSize);
    const bz = Math.floor((bird.position.z + halfSize) / GRID.cellSize);

    // Clamp to valid grid
    const gridX = Math.max(0, Math.min(GRID.gridCount - 1, bx));
    const gridZ = Math.max(0, Math.min(GRID.gridCount - 1, bz));

    const blockCenterX = -halfSize + gridX * GRID.cellSize + GRID.cellSize / 2;
    const blockCenterZ = -halfSize + gridZ * GRID.cellSize + GRID.cellSize / 2;

    // Pick a random sidewalk side
    const side = Math.floor(Math.random() * 4);
    const halfInner = innerSize / 2;
    const swOffset = GRID.sidewalkWidth / 2;

    let px, pz;
    if (side === 0) {
        px = blockCenterX + (Math.random() - 0.5) * (innerSize - 2);
        pz = blockCenterZ - halfInner + swOffset;
    } else if (side === 1) {
        px = blockCenterX + (Math.random() - 0.5) * (innerSize - 2);
        pz = blockCenterZ + halfInner - swOffset;
    } else if (side === 2) {
        px = blockCenterX - halfInner + swOffset;
        pz = blockCenterZ + (Math.random() - 0.5) * (innerSize - 2);
    } else {
        px = blockCenterX + halfInner - swOffset;
        pz = blockCenterZ + (Math.random() - 0.5) * (innerSize - 2);
    }

    return new THREE.Vector3(px, 0.2, pz);
}

function checkPedestrianProximity(birdPos) {
    for (let i = 0; i < peds.length; i++) {
        const p = peds[i];
        const dist = birdPos.distanceTo(p.position);
        if (dist < BIRD_CFG.scatterDistance) {
            return true;
        }
    }
    return false;
}
```

```javascript
        // Update each bird
        flock.birds.forEach(bird => {
            // Handle landed state
            if (bird.userData.state === 'landed') {
                // Check for pedestrian proximity - scatter if too close
                if (checkPedestrianProximity(bird.position)) {
                    bird.userData.state = 'takeoff';
                    bird.userData.velocity.set(
                        (Math.random() - 0.5) * 0.2,
                        BIRD_CFG.takeoffSpeed,
                        (Math.random() - 0.5) * 0.2
                    );
                    return;
                }

                // Check if time to take off naturally
                if (t > bird.userData.takeoffTime) {
                    bird.userData.state = 'takeoff';
                    bird.userData.velocity.set(
                        (Math.random() - 0.5) * 0.1,
                        BIRD_CFG.takeoffSpeed,
                        (Math.random() - 0.5) * 0.1
                    );
                }

                // Subtle idle animation while landed
                bird.rotation.y += Math.sin(t * 2 + bird.userData.wingPhase) * 0.01;
                return;
            }

            // Handle takeoff state
            if (bird.userData.state === 'takeoff') {
                bird.position.add(bird.userData.velocity);
                bird.userData.velocity.y *= 0.98; // Slow vertical climb

                // Animate wings rapidly during takeoff
                bird.userData.wingPhase += 0.6;
                const wingAngle = Math.sin(bird.userData.wingPhase) * 0.6;
                if (bird.userData.leftWing) bird.userData.leftWing.rotation.z = wingAngle;
                if (bird.userData.rightWing) bird.userData.rightWing.rotation.z = -wingAngle;

                // Transition to flying once high enough
                if (bird.position.y > BIRD_CFG.minHeight) {
                    bird.userData.state = 'flying';
                    bird.userData.velocity.y = 0;
                }
                return;
            }

            // Handle landing state
            if (bird.userData.state === 'landing') {
                const target = bird.userData.landingTarget;
                const toTarget = target.clone().sub(bird.position);
                const dist = toTarget.length();

                if (dist < 0.3) {
                    // Landed!
                    bird.position.copy(target);
                    bird.userData.state = 'landed';
                    bird.userData.velocity.set(0, 0, 0);
                    bird.userData.takeoffTime = t + BIRD_CFG.minLandTime +
                        Math.random() * (BIRD_CFG.maxLandTime - BIRD_CFG.minLandTime);

                    // Fold wings
                    if (bird.userData.leftWing) bird.userData.leftWing.rotation.z = 0;
                    if (bird.userData.rightWing) bird.userData.rightWing.rotation.z = 0;

                    // Face random direction
                    bird.rotation.y = Math.random() * Math.PI * 2;
                    return;
                }

                // Move toward landing spot
                toTarget.normalize().multiplyScalar(BIRD_CFG.landingSpeed);
                bird.userData.velocity.lerp(toTarget, 0.1);
                bird.position.add(bird.userData.velocity);

                // Face direction of travel
                const lookTarget = bird.position.clone().add(bird.userData.velocity);
                bird.lookAt(lookTarget);

                // Slow wing flapping during descent
                bird.userData.wingPhase += 0.2;
                const wingAngle = Math.sin(bird.userData.wingPhase) * 0.3;
                if (bird.userData.leftWing) bird.userData.leftWing.rotation.z = wingAngle;
                if (bird.userData.rightWing) bird.userData.rightWing.rotation.z = -wingAngle;
                return;
            }

            // Flying state - check if should start landing
            if (bird.userData.state === 'flying' && Math.random() < BIRD_CFG.landingChance) {
                const perch = findPerchSpot(bird);
                if (perch) {
                    bird.userData.state = 'landing';
                    bird.userData.landingTarget = perch;
                    return;
                }
            }

            // Normal flying behavior (boids algorithm)
            // Calculate boids forces
            const separation = new THREE.Vector3();
            const cohesion = new THREE.Vector3();
            const alignment = new THREE.Vector3();
            let separationCount = 0;
            let cohesionCount = 0;
            let alignmentCount = 0;

            // Check against other birds in same flock
            flock.birds.forEach(other => {
                if (other === bird || other.userData.state !== 'flying') return;

                const dist = bird.position.distanceTo(other.position);

                // Separation - steer away from nearby birds
                if (dist < BIRD_CFG.separationDist && dist > 0) {
                    const diff = bird.position.clone().sub(other.position);
                    diff.normalize().divideScalar(dist);
                    separation.add(diff);
                    separationCount++;
                }

                // Cohesion - steer toward center of nearby birds
                if (dist < BIRD_CFG.cohesionDist) {
                    cohesion.add(other.position);
                    cohesionCount++;
                }

                // Alignment - match velocity of nearby birds
                if (dist < BIRD_CFG.alignmentDist) {
                    alignment.add(other.userData.velocity);
                    alignmentCount++;
                }
            });

            // Average and apply weights
            if (separationCount > 0) {
                separation.divideScalar(separationCount);
                separation.normalize().multiplyScalar(BIRD_CFG.separationWeight);
            }

            if (cohesionCount > 0) {
                cohesion.divideScalar(cohesionCount);
                cohesion.sub(bird.position);
                cohesion.normalize().multiplyScalar(BIRD_CFG.cohesionWeight);
            }

            if (alignmentCount > 0) {
                alignment.divideScalar(alignmentCount);
                alignment.normalize().multiplyScalar(BIRD_CFG.alignmentWeight);
            }

            // Waypoint steering
            const toWaypoint = new THREE.Vector3();
            if (flock.targetWaypoint) {
                toWaypoint.subVectors(flock.targetWaypoint, bird.position);
                toWaypoint.normalize().multiplyScalar(0.5);
            }

            // Random wandering
            const wander = new THREE.Vector3(
                (Math.random() - 0.5) * BIRD_CFG.wanderWeight,
                (Math.random() - 0.5) * BIRD_CFG.wanderWeight * 0.3,
                (Math.random() - 0.5) * BIRD_CFG.wanderWeight
            );

            // Height bounds steering
            const heightSteer = new THREE.Vector3();
            if (bird.position.y < BIRD_CFG.minHeight) {
                heightSteer.y = 0.5;
            } else if (bird.position.y > BIRD_CFG.maxHeight) {
                heightSteer.y = -0.5;
            }

            // Combine all forces
            const acceleration = new THREE.Vector3();
            acceleration.add(separation);
            acceleration.add(cohesion);
            acceleration.add(alignment);
            acceleration.add(toWaypoint);
            acceleration.add(wander);
            acceleration.add(heightSteer);

            // Apply acceleration to velocity
            bird.userData.velocity.add(acceleration.multiplyScalar(BIRD_CFG.turnSpeed));

            // Limit speed
            const speed = bird.userData.velocity.length();
            if (speed > BIRD_CFG.speed * 1.5) {
                bird.userData.velocity.normalize().multiplyScalar(BIRD_CFG.speed * 1.5);
            } else if (speed < BIRD_CFG.speed * 0.5) {
                bird.userData.velocity.normalize().multiplyScalar(BIRD_CFG.speed * 0.5);
            }

            // Update position
            bird.position.add(bird.userData.velocity);

            // Face direction of travel
            if (bird.userData.velocity.length() > 0.01) {
                const lookTarget = bird.position.clone().add(bird.userData.velocity);
                bird.lookAt(lookTarget);
            }

            // Animate wings
            bird.userData.wingPhase += 0.4;
            const wingAngle = Math.sin(bird.userData.wingPhase) * 0.5;
            if (bird.userData.leftWing) {
                bird.userData.leftWing.rotation.z = wingAngle;
            }
            if (bird.userData.rightWing) {
                bird.userData.rightWing.rotation.z = -wingAngle;
            }
        });
```
